## Riešenie: Nezavádzať nové systémy, ZRFERENCOVAŤ EXISTUJÚCE

### 1. Upgrade MetricAuthorityGuard (1 deň)

**Cieľ:** Posilniť existujúci guard na enforce.

**Implementácia:**
```javascript
export class MetricAuthorityGuard {
  constructor() {
    this.strictMode = false;  // Default: warn only
    this.readOnlyMode = false;  // Legacy fields read-only
    this.allowedWriters = new Map();
  }

  registerWriter(systemName, allowedMetrics) {
    this.allowedWriters.set(systemName, new Set(allowedMetrics));
  }

  setStrictMode(strict) {
    this.strictMode = strict;
  }

  setReadOnlyMode(readOnly) {
    this.readOnlyMode = readOnly;
  }

  assertMetricAuthority(systemName, metric) {
    if (!metric || !systemName) return;

    const allowedList = this.allowedWriters.get(systemName);
    if (!allowedList) {
      if (this.strictMode) {
        throw new Error(`[MetricAuthorityGuard] Unauthorized writer: ${systemName}`);
      }
      console.warn(`[MetricAuthorityGuard] Unauthorized writer (warn only): ${systemName} → ${metric}`);
      return;
    }

    if (!allowedList.has(metric)) {
      if (this.strictMode) {
        throw new Error(`[MetricAuthorityGuard] Unauthorized metric: ${systemName} → ${metric}`);
      }
      console.warn(`[MetricAuthorityGuard] Unauthorized metric (warn only): ${systemName} → ${metric}`);
      return;
    }
  }

  checkLegacyRead(field) {
    if (this.readOnlyMode) {
      throw new Error(`[SemanticMetricAdapter] Legacy field read (read-only mode): ${field}`);
    }
    console.warn(`[SemanticMetricAdapter] ⚠️ LEGACY FIELD: ${field}`);
  }
}

// Global instance
export const metricAuthorityGuard = new MetricAuthorityGuard();

// Registrovaní writeri
metricAuthorityGuard.registerWriter('NodeMetricEngine', ['synergy', 'harmony', 'stability', 'corruption', 'loadPressure']);
metricAuthorityGuard.registerWriter('HarmonyStabilizationSystem_v1', ['harmony']);
metricAuthorityGuard.registerWriter('HarmonicHealingVisualSystem_Session134', ['stability', 'corruption']);
metricAuthorityGuard.registerWriter('LinkCorruptionTransmission_v1', ['corruption']);
```

**Použitie:**
```javascript
// V SemanticMetricAdapter
export function getNodeCanonicalMetrics(node) {
  const metrics = node?.userData?.metrics || {};

  // ✅ Canonical path
  const harmony = firstDefined(metrics.harmony, metrics.harmonyNorm);

  // ❌ Legacy path - warn or error v readOnly mode
  if (typeof node?.userData?.harmonyLevel !== 'undefined') {
    metricAuthorityGuard.checkLegacyRead('node.userData.harmonyLevel');
    return harmony;  // použiš canonical
  }

  return {
    synergy,
    harmony,  // použiš canonical
    // ...
  };
}

// V HarmonyStabilizationSystem
export function setNodeHarmony(node, level) {
  metricAuthorityGuard.assertMetricAuthority('HarmonyStabilizationSystem_v1', 'harmony');
  node.userData.metrics.harmony = level;  // ✅ Canonical
}
```
