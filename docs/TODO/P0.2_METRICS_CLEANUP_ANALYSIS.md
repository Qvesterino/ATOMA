## Navrhované Riešenie: Metric Writer Registry

### 1. Metric Writer Registry (MWR)

**Cieľ:** Centralizovaný register všetkých metrických writierov s jasným oprávnením.

**Štruktúra:**
```javascript
// src/metrics/MetricWriterRegistry.js

export class MetricWriterRegistry {
    constructor() {
        this.writers = new Map();
        this.writeLock = false;
        this.auditLog = [];
    }

    /**
     * Register a metric writer
     * @param {string} writerId - Unique identifier (e.g., 'HarmonyStabilizationSystem_v1')
     * @param {Set<string>} allowedMetrics - Set of metric names this writer can modify
     * @param {Object} options - { strict: boolean, allowList: boolean, audit: boolean }
     */
    registerWriter(writerId, allowedMetrics, options = {}) {
        this.writers.set(writerId, {
            id: writerId,
            allowedMetrics: new Set(allowedMetrics),
            strict: options.strict ?? false,
            allowList: options.allowList ?? false,
            audit: options.audit ?? true,
            writeCount: 0,
            lastWriteAt: null
        });
    }

    /**
     * Check if a writer is allowed to write a metric
     */
    canWrite(writerId, metricName) {
        const writer = this.writers.get(writerId);
        if (!writer) {
            this._logAudit('UNREGISTERED_WRITER', { writerId, metricName });
            return false;
        }

        const allowed = writer.allowedMetrics.has(metricName);
        if (!allowed) {
            this._logAudit('UNAUTHORIZED_WRITE', { writerId, metricName });
        }

        return allowed;
    }

    /**
     * Record a write operation (for audit)
     */
    recordWrite(writerId, metricName, value, nodeId, linkId) {
        const writer = this.writers.get(writerId);
        if (!writer) return;

        writer.writeCount++;
        writer.lastWriteAt = Date.now();

        if (writer.audit) {
            this._logAudit('WRITE', { writerId, metricName, value, nodeId, linkId });
        }
    }

    /**
     * Enable/disable write lock (for debugging/migration)
     */
    setWriteLock(locked) {
        this.writeLock = locked;
    }

    /**
     * Get audit log
     */
    getAuditLog(limit = 100) {
        return this.auditLog.slice(-limit);
    }

    _logAudit(type, data) {
        this.auditLog.push({
            type,
            timestamp: Date.now(),
            ...data
        });

        // Keep audit log size manageable
        if (this.auditLog.length > 1000) {
            this.auditLog.shift();
        }
    }
}

// Global instance
export const metricWriterRegistry = new MetricWriterRegistry();
```

**Použitie v NodeMetricEngine:**
```javascript
import { metricWriterRegistry } from './MetricWriterRegistry.js';

// Initialize canonical writers
metricWriterRegistry.registerWriter('NodeMetricEngine.js', 
    ['synergy', 'harmony', 'stability', 'corruption', 'loadPressure'],
    { strict: true, audit: true }
);
```

**Použitie pri zápise:**
```javascript
// V akomkoľvek systéme, ktorý píše metriky
if (metricWriterRegistry.canWrite('HarmonyStabilizationSystem_v1', 'harmony')) {
    node.userData.metrics.harmony = newValue;
    metricWriterRegistry.recordWrite('HarmonyStabilizationSystem_v1', 'harmony', newValue, nodeId);
} else {
    console.warn('[MetricWriterRegistry] Unauthorized write attempt', {
        writer: 'HarmonyStabilizationSystem_v1',
        metric: 'harmony',
        nodeId
    });
}
```

### 2. Mirror Field Compatibility Bridge

**Cieľ:** Zjednodušiť migráciu z mirror fields na canonical paths.

**Štruktúra:**
```javascript
// src/metrics/MetricCompatibilityBridge.js

export class MetricCompatibilityBridge {
    constructor() {
        this.mirrorFields = {
            node: new Map([
                ['harmonyLevel', 'metrics.harmony'],
                // Ďalšie mirror fields...
            ]),
            link: new Map([
                ['harmonyLevel', 'metrics.harmony'],
                // Ďalšie mirror fields...
            ])
        };

        this.readOnlyMode = false;
    }

    /**
     * Get metric value with mirror fallback (read-only)
     */
    getMetric(object, mirrorKey, fallback = 0) {
        // Prefer canonical path
        const canonicalPath = this.mirrorFields.node.get(mirrorKey) 
                           || this.mirrorFields.link.get(mirrorKey);

        if (canonicalPath) {
            const value = this._getNested(object, canonicalPath);
            if (value !== undefined) {
                return value;
            }
        }

        // Fallback to mirror (read-only warning)
        const mirrorValue = object.userData[mirrorKey];
        if (mirrorValue !== undefined && !this.readOnlyMode) {
            console.warn(`[MetricCompatibilityBridge] Reading deprecated mirror field: ${mirrorKey}`);
            return mirrorValue;
        }

        return fallback;
    }

    /**
     * Set metric value (canonical path only)
     */
    setMetric(object, canonicalKey, value, writerId) {
        // Validate write authority
        if (globalThis.metricWriterRegistry) {
            if (!globalThis.metricWriterRegistry.canWrite(writerId, canonicalKey)) {
                console.error('[MetricCompatibilityBridge] Unauthorized write attempt', {
                    writer: writerId,
                    metric: canonicalKey
                });
                return false;
            }
        }

        // Write to canonical path
        if (!object.userData) {
            object.userData = {};
        }

        if (!object.userData.metrics) {
            object.userData.metrics = {};
        }

        object.userData.metrics[canonicalKey] = value;

        // Record write
        if (globalThis.metricWriterRegistry) {
            const nodeId = object.userData?.nodeId || object.id;
            globalThis.metricWriterRegistry.recordWrite(writerId, canonicalKey, value, nodeId);
        }

        return true;
    }

    /**
     * Enable read-only mode (for migration)
     */
    setReadOnlyMode(readOnly) {
        this.readOnlyMode = readOnly;
    }

    _getNested(obj, path) {
        return path.split('.').reduce((current, key) => current?.[key], obj);
    }
}

// Global instance
export const metricCompatibilityBridge = new MetricCompatibilityBridge();
```

**Použitie:**
```javascript
// Namiesto priameho čítania/zápisu
const harmony = metricCompatibilityBridge.getMetric(node, 'harmonyLevel', 0.5);
metricCompatibilityBridge.setMetric(node, 'harmony', 0.75, 'HarmonyStabilizationSystem_v1');
```

### 2. Runtime Audit Check

**V MetricsRuntime_v1:**
```javascript
// Runtime audit pre non-canonical writes (5-sekundový interval)
if (this._runtimeAudit.accumulator >= 5.0) {
    this._auditCanonicalFields();
    this._runtimeAudit.accumulator = 0;
}

_auditCanonicalFields() {
    const now = Date.now();
    const staleMs = 4000;

    // Check node harmony mirror vs canonical
    this.nodes?.forEach(node => {
        if (!node.userData) return;

        const canonicalHarmony = node.userData.metrics?.harmony;
        const mirrorHarmony = node.userData.harmonyLevel;

        if (typeof canonicalHarmony === 'number' && 
            typeof mirrorHarmony === 'number') {
            const diff = Math.abs(canonicalHarmony - mirrorHarmony);
            if (diff > 0.01) {
                console.warn('[MetricsRuntime_v1] Metric mirror desync detected', {
                    nodeId: node.userData.nodeId || node.id,
                    canonical: canonicalHarmony,
                    mirror: mirrorHarmony,
                    diff
                });
            }
        }
    });
}
```
