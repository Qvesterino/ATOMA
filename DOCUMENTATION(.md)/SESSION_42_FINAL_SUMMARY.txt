    isCritical: boolean,          // Critical flag
    updatedAt: timestamp          // Last update (ms)
  };

LINK VISUAL METRICS:
  link.userData.visualMetrics = {
    qualityNorm: 0.0–1.0,        // From LinkQualityCalculator
    stressNorm: 0.0–1.0,         // Inverse of structural quality
    updatedAt: timestamp          // Last update (ms)
  };

================================================================================
NORMALIZATION RULES
================================================================================

RULE 1: From 0–100 Scale
  normalized = Math.max(0, Math.min(1, raw / 100))
  
  Examples:
    stability: 85 → stabilityNorm: 0.85
    quality: 100 → qualityNorm: 1.0

RULE 2: Already 0–1 Scale
  normalized = Math.max(0, Math.min(1, raw))
  
  Examples:
    energyNorm: 0.75 → energyNorm: 0.75
    loadRatio: 0.5 → loadNorm: 0.5

RULE 3: Computed Values
  synergyNorm = (harmonyNorm * 0.6 + stabilityNorm * 0.4)
  stressNorm = (100 - structuralScore) / 100

RULE 4: Safety
  All values clamped to [0, 1]
  NaN/Infinity replaced with safe defaults
  No value can ever be outside [0, 1]
