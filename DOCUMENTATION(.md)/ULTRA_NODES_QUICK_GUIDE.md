```javascript
// Core sizes
const coreAGeometry = new THREE.SphereGeometry(0.15, 16, 16);  // Heart
const coreBGeometry = new THREE.IcosahedronGeometry(0.4, 3);   // Sphere
const coreCGeometry = new THREE.SphereGeometry(0.6, 16, 16);   // Shell

// Ring count
const ringCount = isSpecial ? 3 : (Math.random() < 0.5 ? 2 : 1);

// Particle count
const particleCount = isSpecial ? 12 : 8;

// Glow size
const outerGlowGeometry = new THREE.IcosahedronGeometry(1.2, 4);  // 1.2 = size
```

```javascript
// Glow brightness
const glowPulse = (0.4 + Math.sin(time * 2) * 0.15) * glowBreathing;  // 0.4 = base
```
