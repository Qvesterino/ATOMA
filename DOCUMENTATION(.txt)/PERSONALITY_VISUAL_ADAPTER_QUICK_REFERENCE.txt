FORMULA REFERENCE
=================

clarityBoost
─────────────
= (harmonyNorm × 0.40) + (stabilityNorm × 0.30) + (qualityNorm × 0.30)
→ Intellectual focus, coherence, quality
→ Use for: sharpen effects, increase glow clarity

resonanceBoost
───────────────
= (avgSurroundingSynergy × 0.70) + (harmonyNorm × 0.30)
  [or just harmonyNorm if no links]
→ Connection harmony, link resonance
→ Use for: pulse frequency, link animation, orbiter patterns

entropyPenalty
───────────────
= (corruptionNorm × 0.50) + ((1 - stabilityNorm) × 0.30) + (loadNorm × 0.20)
→ Chaos, disorder, degradation
→ Use for: glitch effects, scatter, noise, distortion

focusShift
───────────
= ((1 - stabilityNorm) × 0.60) + (loadNorm × 0.40)
→ Overload, scattered attention
→ Use for: jitter, erratic motion, particle scatter

corruptionSignal
─────────────────
= corruptionNorm (direct pass-through)
→ Corruption level, degradation
→ Use for: red tint, decay effects, damage visualization

EXAMPLE VFX INTEGRATION
==========================
function updateNodeVFX(node, deltaTime) {
  const pv = node.userData.personalityVisual;
  if (!pv) return;
  
  // Clarity: sharpen and brighten
  node.material.emissiveIntensity = 0.5 + (pv.clarityBoost * 0.5);
  
  // Resonance: speed up rotation
  node.rotationSpeed = 1.0 + (pv.resonanceBoost * 2.0);
  
  // Entropy: add particles
  if (pv.entropyPenalty > 0.3) {
    spawnChaosParticles(node, pv.entropyPenalty);
  }
  
  // Focus: jitter position
  const jitter = pv.focusShift * 0.02;
  node.position.add(new THREE.Vector3(
    (Math.random() - 0.5) * jitter,
    (Math.random() - 0.5) * jitter,
    (Math.random() - 0.5) * jitter
  ));
  
  // Corruption: red tint
  node.material.color.lerp(
    new THREE.Color(1, 0.3, 0.2),
    pv.corruptionSignal * 0.3
  );
}

  node.material.emissiveIntensity,
  targetIntensity,

─────────────────────────────────

  personalityVis.clarityBoost * 0.4 +      // Clarity helps
