Example: Apply glow intensities to link materials

// Simple approach: Update materials with glow values
function updateLinkMaterials(links, glowEngine) {
  for (const link of links) {
    if (!link || !link.material) continue;
    
    const glow = link.userData?.visualGlow;
    if (!glow) continue;
    
    // Apply glow to emissive intensity
    link.material.emissiveIntensity = glow.glowIntensity * 1.5;
    
    // Adjust opacity
    link.material.opacity = glow.glowIntensity;
    
    // Optional: Add chaos pulse as animation
    if (glow.corruptionPulse > 0) {
      const pulse = Math.sin(Date.now() * 0.003) * glow.corruptionPulse;
      link.material.emissiveIntensity += pulse;
    }
  }
}

BLOCK 5: QUALITY-BASED COLOR AND GLOW
================================================================================

Example: Color links based on quality, glow based on synergy

function colorLinkByQuality(links) {
  for (const link of links) {
    const glow = link.userData?.visualGlow;
    if (!glow || !link.material) continue;
    
    // Color based on quality
    let color;
    if (glow.qualityNorm > 0.75) {
      color = 0x00ff00;  // Green: high quality
    } else if (glow.qualityNorm > 0.50) {
      color = 0x00ffff;  // Cyan: medium quality
    } else if (glow.qualityNorm > 0.25) {
      color = 0xffff00;  // Yellow: low quality
    } else {
      color = 0xff0000;  // Red: critical
    }
    
    // Apply color
    if (link.material.color) {
      link.material.color.setHex(color);
    }
    if (link.material.emissive) {
      link.material.emissive.setHex(color);
    }
    
    // Glow intensity based on synergy
    link.material.emissiveIntensity = glow.glowIntensity * 2.0;
  }
}

BLOCK 6: DETECT AND HIGHLIGHT CHAOTIC LINKS
================================================================================

Example: Find links with high corruption pulse

function highlightChaoticLinks(links) {
  const chaoticLinks = [];
  
  for (const link of links) {
    const glow = link.userData?.visualGlow;
    if (!glow) continue;
    
    // Links with corruption pulse > 0.3 are chaotic
    if (glow.corruptionPulse > 0.3) {
      chaoticLinks.push({
        link,
        chaosLevel: glow.corruptionPulse,
        corruption: (glow.corruptionPulse / 0.5) * 100  // Convert to %
      });
    }
  }
  
  // Optional: Log report
  if (chaoticLinks.length > 0) {
    console.log(`[Network] Found ${chaoticLinks.length} chaotic links:`);
    chaoticLinks.forEach(({ link, corruption }) => {
      console.log(`  Link ${link.id}: ${corruption.toFixed(1)}% chaos`);
    });
  }
  
  return chaoticLinks;
}

function monitorGlowPerformance(glowEngine) {
  const stats = glowEngine.getStats();
  
  console.group('[Glow Engine Performance]');
  console.log(`  Computes: ${stats.computeCount}`);
  console.log(`  Visual metrics used: ${stats.visualMetricsUsagePercent}%`);
  console.log(`  Fallbacks used: ${stats.fallbackUsagePercent}%`);
  console.log(`  Average time: ${stats.averageTimeMs.toFixed(3)}ms`);
  console.log(`  Total time: ${stats.totalTimeMs.toFixed(2)}ms`);
  console.groupEnd();
  
  // Warn if performance degrading
  if (stats.averageTimeMs > 0.5) {
    console.warn('⚠ Glow engine performance degrading!');
  }
}

BLOCK 9: VERIFY INTEGRATION ON STARTUP

Example: Verify glow engine is working

function verifyGlowEngineIntegration(game) {
  console.group('[Verify] LinkGlowSynergyEngine_v2 Integration');
  
  if (!game.glowEngine_v2) {
    console.error('❌ glowEngine_v2 not initialized');
    console.groupEnd();
    return false;
  }
  
  if (!game.nodeLinkingSystem.links || game.nodeLinkingSystem.links.length === 0) {
    console.warn('⚠ No links to test');
    console.groupEnd();
    return true;
  }
  
  // Test with first link
  const testLink = game.nodeLinkingSystem.links[0];
  try {
    game.glowEngine_v2.update([testLink], 0.016);
    
    const glow = testLink.userData.visualGlow;
    if (!glow) {
      console.error('❌ No glow profile generated');
      console.groupEnd();
      return false;
    }
    
    console.log('✓ Glow profile generated');
    console.log('  Intensity:', glow.glowIntensity.toFixed(3));
    console.log('  Synergy:', glow.synergyNorm.toFixed(3));
    console.log('  Quality:', glow.qualityNorm.toFixed(3));
    console.log('  Method:', glow.debug.method);
    
    console.groupEnd();
    return true;
    
  } catch (err) {
    console.error('❌ Update failed:', err);
    console.groupEnd();
    return false;
  }
}
