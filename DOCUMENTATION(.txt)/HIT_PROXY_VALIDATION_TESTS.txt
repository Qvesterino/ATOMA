Test 3.2: Continuous Position Sync
────────────────────────────────────────────────────────────────────────────

Command:
  // Test position update over 5 frames
  const node = game.aiNodes.nodes[0];
  const nodeId = node.userData.nodeId;
  const initialPos = node.position.clone();
  
  // Manually move node
  node.position.x += 5;
  
  // Call sync
  window.hitProxySystem.update(0.016);
  
  const proxy = window.hitProxySystem.registry.getProxy(nodeId);
  console.log({
    nodeMoved: !node.position.equals(initialPos),
    proxyFollowed: proxy.position.equals(node.position)
  });

Expected:
  {
    nodeMoved: true,
    proxyFollowed: true
  }

Failure Indicators:
  proxyFollowed: false (sync failed)


Test 3.3: Position Array Consistency
────────────────────────────────────────────────────────────────────────────

  let unsyncedCount = 0;

Test 6.1: Real Visual Protection

Command:

  let totalCores = 0;

      totalCores++;
      if (obj.raycast && obj.raycast.toString().includes('no-op')) {
        coreRaycastDisabled++;
      }
    }
  });
  
  console.log(`Protected cores: ${coreRaycastDisabled}/${totalCores}`);

Expected:
  Protected cores: 15/15 (or matching total)

Failure Indicators:
  Protected cores < total


Test 6.2: No Proxy Intersection on Real Visuals
────────────────────────────────────────────────────────────────────────────

Command:
  const raycaster = window.safeProxyRaycaster;
  raycaster.setFromCamera({ x: 0, y: 0 }, window.game.camera);
  
  // Get all objects in scene (not just proxies)
  let allObjects = [];
  game.scene.traverse(obj => {
    if (obj instanceof THREE.Mesh) allObjects.push(obj);
  });
  
  // Raycast against all objects
  const allHits = raycaster.intersectObjects(allObjects);
  
  // Should only hit proxies
  let proxyHits = allHits.filter(h => h.object.userData.isHitProxy).length;
  let nonProxyHits = allHits.filter(h => !h.object.userData.isHitProxy).length;
  
  console.log({
    proxyHits: proxyHits,
    nonProxyHits: nonProxyHits,
    allAreProxies: nonProxyHits === 0
  });

Expected:
  {
    proxyHits: 1,
    nonProxyHits: 0,
    allAreProxies: true
  }

Failure Indicators:
  nonProxyHits > 0 (real visuals were hit)


Test 6.3: No Geometry Mutation
────────────────────────────────────────────────────────────────────────────

Command:
  // Check no frozen geometry accessed
  const proxy = HitProxyDebug.allProxies()[0];
