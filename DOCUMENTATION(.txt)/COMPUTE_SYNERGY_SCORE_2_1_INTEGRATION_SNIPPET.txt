SNIPPET 3: UPDATE SYNERGY SCORES IN GAME LOOP
================================================================================

Add to AtomaGame.update(dt) method (in the link update section):

// ========== UPDATE SYNERGY SCORES WITH 2.1 ==========
// Call this after VisualMetricModel.update(dt) and before rendering
if (this.synergyCalculator2_1 && this.nodeLinkingSystem.links) {
  for (const link of this.nodeLinkingSystem.links) {
    if (!link) continue;
    
    try {
      // Compute synergy using 2.1 (falls back to 2.0 if needed)
      const synergyResult = this.synergyCalculator2_1.compute(link, {
        linkingSystem: this.nodeLinkingSystem,
        correlationEngine: this.linkCorrelationEngine,
        priorityHistoryEngine: this.priorityHistoryEngine,
        priorityDecayEngine: this.priorityDecayEngine,
        config: {}
      });
      
      // Store result for VFX systems to read
      if (!link.userData) link.userData = {};
      link.userData.synergy2_1 = synergyResult;
      
    } catch (err) {
      console.error('[AtomaGame] Synergy2_1 compute error:', err);
    }
  }
}

SNIPPET 6: VERIFY 2.1 INTEGRATION ON STARTUP
================================================================================

Add to your game initialization (after all systems created):

// Verify ComputeSynergyScore2_1 is working
function verifyComputeSynergy2_1(game) {
  console.group('[Verify] ComputeSynergyScore2_1 Integration');
  
  if (!game.synergyCalculator2_1) {
    console.error('❌ synergyCalculator2_1 not initialized');
    console.groupEnd();
    return false;
  }
  
  if (!game.nodeLinkingSystem.links || game.nodeLinkingSystem.links.length === 0) {
    console.warn('⚠ No links to test');
    console.groupEnd();
    return true;  // Not a failure, just nothing to test
  }
  
  // Test with first link
  const testLink = game.nodeLinkingSystem.links[0];
  try {
    const result = game.synergyCalculator2_1.compute(testLink, {
      linkingSystem: game.nodeLinkingSystem,
      correlationEngine: game.linkCorrelationEngine,
    });
    
    console.log('✓ Compute successful');
    console.log('  Result:', result);
    
    if (result.debug.method === 'visual') {
      console.log('✓ Visual metrics being used');
    } else if (result.debug.method === 'legacy') {
      console.log('ℹ Falling back to legacy 2.0 (visualMetrics unavailable)');
    }
    
    console.log(`✓ synergyNorm: ${result.synergyNorm.toFixed(3)}`);
    console.groupEnd();
    return true;
    
  } catch (err) {
    console.error('❌ Compute failed:', err);
    console.groupEnd();
    return false;
  }
}

SNIPPET 7: INTEGRATE WITH LINKGLOWSYNERGYENGINE (WEEK 2)
================================================================================

Preview of how LinkGlowSynergyEngine will use synergyNorm in Week 2:

// In LinkGlowSynergyEngine1_0 (Week 2 upgrade):

function updateLinkGlow(link, game) {
  // Get new synergy result from 2.1
  const synergy2_1 = link.userData?.synergy2_1;
  
  if (synergy2_1) {
    // Use clean synergyNorm for glow intensity
    const glowIntensity = synergy2_1.synergyNorm * 2.0;  // 0–2 range
    
    // Map to color based on tier
    const tierToColor = {
      'low': 0x0088ff,       // Blue
      'medium': 0x00ff88,    // Cyan
      'high': 0xffff00,      // Yellow
      'critical': 0xff0000   // Red
    };
    
    link.material.emissive.setHex(tierToColor[synergy2_1.tier] || 0x0088ff);
    link.material.emissiveIntensity = glowIntensity;
  }
}
