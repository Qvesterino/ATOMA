SPREAD MECHANICS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Direction:
  ├─ Unilateral:    Source (0.75) → Target (0.20)
  ├─ Bidirectional: Source (0.70) ↔ Target (0.68)
  └─ Saturation:    Spreads slow when differential < 0.1

Speed:
  ├─ Base rate: 0.15 corruption/second
  ├─ Modified by: (source-target)^1.5 (exponential curve)
  ├─ Resistance: 1/linkResistance × 1/nodeImmunity
  └─ Multi-source: ×1.15-1.5 boost (proximity effect)

Saturation:
  └─ When differential < 0.1: Damping applied (slows spread)

CUSTOMIZATION
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Per-Link Resistance:
  linkingSystem.setLinkContagionResistance(link, resistance)
  Range: 0.1 (fastest) to 10.0 (slowest)
  Default: 0.8 (20% slower than baseline)
  
  Example: 2.0 = 2x slower spread
  Example: 0.5 = 2x faster spread

Per-Node Immunity:
  linkingSystem.setNodeContagionResistance(node, immunity)
  Range: 0.1 (most vulnerable) to 10.0 (most resistant)
  Default: 1.0 (normal infection rate)
  
  Example: 2.0 = 2x harder to infect
  Example: 0.5 = 2x easier to infect


QUERIES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Link Status:
  status = linkingSystem.getLinkContagionStatus(link)
  Returns: { isInfected, intensity (0-1), direction }
  
  direction values:
    'source->target'   (unilateral from source to target)
    'target->source'   (unilateral from target to source)
    'bidirectional'    (both spreading simultaneously)
    null               (no contagion)

Node Threat:
  threat = linkingSystem.getNodeContagionThreat(node)
  Returns: { threatLevel (0-1), sourceCount, speed }
  
  threatLevel:  0-1 composite threat (up to 1.0 at 4+ sources)
  sourceCount:  Number of corrupted neighbors
  speed:        Average infection speed from sources
