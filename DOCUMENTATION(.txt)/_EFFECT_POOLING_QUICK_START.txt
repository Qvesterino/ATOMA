BASIC USAGE

// Create effect with pooling (validates inputs)
const success = orchestrator.addPooled('linkPulse', {
  pulse: pulseMesh,                              // Required: Mesh to animate
  startPos: sourceNode.position.clone(),         // Required: Vector3 snapshot
  endPos: targetNode.position.clone(),           // Required: Vector3 snapshot
  duration: 0.5                                  // Required: seconds
});

if (!success) {
  console.warn('Link pulse rejected (invalid inputs)');
}

───────────────────────────────────────────────────────────────────────────────

EFFECT TYPES

linkPulse
  inputs: pulse, startPos, endPos, duration
  behavior: Pulse travels from start to end over duration

linkPulseRemoval
  inputs: pulse, sourcePos, targetPos, duration
  behavior: Pulse travels from target back to source

shatterParticle
  inputs: particle, velocity, duration
  behavior: Particle moves with velocity, fades

errorFeedback
  inputs: pulse, errorType, duration
  behavior: Error pulse at node (conflict/incompatible)

activationPulse
  inputs: pulse, duration
  behavior: Pulse grows and fades at node location

materialization
  inputs: node, duration
  behavior: Node scales up and fades in

───────────────────────────────────────────────────────────────────────────────

CRITICAL RULES

✅ DO: Clone all Vector3 positions
   startPos: sourceNode.position.clone()

❌ DON'T: Store live references
   startPos: sourceNode  // ← WRONG

✅ DO: Use immutable snapshots
   { velocity: velocity.clone() }

❌ DON'T: Store live node references
   { sourceNode: sourceNode }  // ← WRONG

✅ DO: Check if addPooled succeeded
   if (!orchestrator.addPooled(...)) { handle error }

❌ DON'T: Assume effect was created
   orchestrator.addPooled(...);  // Silent failures possible

EXAMPLE: CREATE LINK EFFECTS

  pulse: pulseMesh,
  startPos: sourceNode.position.clone(),
  endPos: targetNode.position.clone(),
  duration: 0.5
});

// When unlinking
const effect2 = orchestrator.addPooled('linkPulseRemoval', {
  pulse: pulseMesh,
  sourcePos: sourceNode.position.clone(),
  targetPos: targetNode.position.clone(),
  duration: 0.4
});

// Both effects safe even if nodes despawn (uses snapshots)
