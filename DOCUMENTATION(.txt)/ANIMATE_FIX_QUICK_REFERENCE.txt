  - Canonical geometries' materials are frozen
  - animate() tries to modify frozen materials
  - TypeError or silent fail on frozen objects

  const isFrozen = Object.isFrozen(child.material);

Guard 2: Check immutable flag

Guard 3: Skip if either condition true
  if (isFrozen || isImmutable) return;

Guard 4: Try-catch for safety
  try { ... } catch(err) { /* skip */ }
