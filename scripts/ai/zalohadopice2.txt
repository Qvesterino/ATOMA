function Get-Section {
    param(
        [string]$Text,
        [string]$Header
    )

    $pattern = [regex]::Escape($Header) + "\s*([\s\S]*?)(\n## |\z)"
    $m = [regex]::Match($Text, $pattern)
    if ($m.Success) { return $m.Groups[1].Value.Trim() }
    return ""
}

function Get-Section {
    param(
        $text,
        $header
    )

  # podporí formáty:
  # "## VERDICT" ... (sekcia)
  # alebo "VERDICT:" na jednej/viac riadkoch
  $h = $header.Trim()
  $pat1 = [regex]::Escape($h) + "\s*([\s\S]*?)(\r?\n## |\z)"
  $m1 = [regex]::Match($text, $pat1, [System.Text.RegularExpressions.RegexOptions]::IgnoreCase)
  if ($m1.Success) { return $m1.Groups[1].Value.Trim() }

  # fallback: "VERDICT:" / "EXECUTOR_PROMPT:" bloky
  $key = $h -replace '^\#\#\s*',''
  $pat2 = "(?im)^\s*$([regex]::Escape($key))\s*:\s*([\s\S]*?)(\r?\n\s*[A-Z0-9_ -]+\s*:|\z)"
  $m2 = [regex]::Match($text, $pat2)
  if ($m2.Success) { return $m2.Groups[2].Value.Trim() }

  return ""
}
 
 function Get-Verdict([string]$reviewText) {
  $v = Get-Section $reviewText "## VERDICT"
  $v = Get-Section $reviewText "VERDICT"
   if ($v -match "PASS") { return "PASS" }
   if ($v -match "WARN") { return "WARN" }
   if ($v -match "FAIL") { return "FAIL" }
   return "UNKNOWN"
 }

function Invoke-ContinueHeadless([string]$prompt, [string]$configPath) {
  $cfgArg = @()
  if ($configPath) { $cfgArg = @("--config", $configPath) }

  # skús postupne: cn, continue, npx
  if (Get-Command cn -ErrorAction SilentlyContinue) {
    return (& cn @cfgArg -p $prompt 2>&1 | Out-String)
  }
  if (Get-Command continue -ErrorAction SilentlyContinue) {
    return (& continue @cfgArg -p $prompt 2>&1 | Out-String)
  }

  if (Get-Command npx -ErrorAction SilentlyContinue) {
    return (& npx -y @continuedev/cli @cfgArg -p $prompt 2>&1 | Out-String)
  }

  throw "Chýba Continue CLI. Nainštaluj: npm i -g @continuedev/cli  (alebo musí existovať 'npx')."
}
